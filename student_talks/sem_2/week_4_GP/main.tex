\begin{frame}{Постановка задачи}

{\small
\textbf{Цель}

Разработать вычислительно эффективный метод регрессии, устойчивый к выбросам в данных.

\vspace{5pt}

\textbf{Проблема}

Гауссовские процессы (GP) --- популярный инструмент регрессии, но имеют два ключевых недостатка:
\begin{itemize}
    \item \textbf{Вычислительная сложность:} наивная GP-регрессия требует $O(n^3)$ операций из-за инверсии матрицы $K^{-1}$.
    \item \textbf{Неустойчивость к выбросам:} предсказательная дисперсия не зависит от значений наблюдений:
    \[
        V_{\mathrm{GP}}[f(x_*)] = k(x_*,x_*) - k_*^T K^{-1} k_*
    \]
\end{itemize}

\vspace{5pt}

\textbf{Цель работы}

Предложить алгоритм на основе процессов Стьюдента-$t$ (TP) со сложностью $O(n)$, сохраняющий аналитическое решение и обеспечивающий устойчивость к выбросам.
}

\end{frame}

%----------------------------------------------------------------------------------------------------------

\begin{frame}{Гауссовский процесс}

{\small
\textbf{Определение}

Процесс $f(x) \sim \mathcal{GP}(\mu(x), k(x,x'))$, если для любой конечной выборки:
\[
    (f(x_1), \ldots, f(x_n))^T \sim \mathcal{N}(\boldsymbol{\mu}, K), \quad K_{ij} = k(x_i, x_j)
\]

\vspace{3pt}

\textbf{GP-регрессия: предсказание}

\[
    E_{\mathrm{GP}}[f(x_*)] = k_*^T K^{-1} y
\]
\[
    V_{\mathrm{GP}}[f(x_*)] = k(x_*,x_*) - k_*^T K^{-1} k_*
\]

\vspace{3pt}

\textbf{Ключевое ограничение}

\begin{block}{Проблема GP}
Дисперсия $V_{\mathrm{GP}}$ зависит \textbf{только от входных точек} $x_i$, но не от значений наблюдений $y_i$. При наличии выбросов неопределённость \textbf{не увеличивается}.
\end{block}
}

\end{frame}

%----------------------------------------------------------------------------------------------------------

\begin{frame}{Распределение Стьюдента-$t$}

{\small
\textbf{Определение 2.1.} Случайная переменная $y \in \mathbb{R}^n$ имеет многомерное распределение Стьюдента-$t$, $y \sim \mathrm{MVT}(\mu, K, \nu)$, если:

\[
p(y|\mu,K,\nu) = \frac{\Gamma\!\left(\frac{\nu+n}{2}\right)}{\Gamma\!\left(\frac{\nu}{2}\right)\!\left((\nu-2)\pi\right)^{n/2}|K|^{1/2}}
\left(1 + \frac{(y-\mu)^T K^{-1}(y-\mu)}{\nu-2}\right)^{-\frac{\nu+n}{2}}
\]

При $\nu \to \infty$ получаем $\mathcal{N}(y\,|\,\mu, K)$.

\vspace{5pt}

\textbf{Лемма 2.2 (смесь масштабов).}

\[
    \gamma \sim \mathrm{IG}\!\left(\tfrac{\nu}{2},\, \tfrac{\nu-2}{2}\right), \quad
    y\,|\,\gamma \sim \mathcal{N}(\mu,\, \gamma K)
    \quad \Longrightarrow \quad
    y \sim \mathrm{MVT}(\mu, K, \nu)
\]

\begin{block}{Ключевая идея}
Распределение Стьюдента-$t$ = \textbf{гауссовское} с случайным масштабом $\gamma$.
\end{block}
}

\end{frame}

%----------------------------------------------------------------------------------------------------------

\begin{frame}{Процесс Стьюдента-$t$ (TP)}

{\small
\textbf{Определение 2.4.} Процесс $f(x) \sim \mathrm{TP}(\mu(x), k(x,x'), \nu)$, если для любой конечной выборки:
\[
    (f(x_1), \ldots, f(x_n))^T \sim \mathrm{MVT}(\boldsymbol{\mu}, K, \nu)
\]

\vspace{3pt}

\textbf{Лемма 2.3 (условное распределение).}

Для совместно $t$-распределённых векторов $f_1, f_2$:
\[
    f_1 \,|\, f_2 \sim \mathrm{MVT}(\mu_{1|2},\; K_{1|2},\; \nu_{1|2})
\]
\[
    \mu_{1|2} = K_{12}K_{22}^{-1}(f_2 - \mu_2) + \mu_1
\]
\[
    K_{1|2} = \frac{\nu - 2 + \beta}{\nu - 2 + n_2}
    \Bigl(K_{11} - K_{12}K_{22}^{-1}K_{21}\Bigr), \quad
    \beta = (f_2-\mu_2)^T K_{22}^{-1}(f_2-\mu_2)
\]
\[
    \nu_{1|2} = \nu + n_2
\]

\begin{block}{Отличие от GP}
Масштабирующий множитель $\dfrac{\nu-2+\beta}{\nu-2+n_2}$ \textbf{зависит от наблюдений} через $\beta$.
\end{block}
}

\end{frame}

%----------------------------------------------------------------------------------------------------------

\begin{frame}{TP-регрессия: переход к апостериорному распределению}

{\footnotesize

\textbf{Шаг 1 — Априорная модель и совместное распределение.}

При $f \sim \mathrm{TP}(0,\, k_\theta,\, \nu)$ по определению процесса:
$$
\begin{pmatrix} f(x_*) \\ \mathbf{y} \end{pmatrix}
\sim
\mathrm{MVT}\!\left(
\mathbf{0},\;
\begin{pmatrix}
  k_\theta(x_*,x_*) & k_*^T \\
  k_*               & K
\end{pmatrix},\;
\nu
\right),
$$
где $[K]_{ij} = k_\theta(x_i, x_j) + \sigma_n^2\delta_{ij}$
и $[k_*]_i = k_\theta(x_*, x_i)$.

\vspace{4pt}

\textbf{Шаг 2 — Применяем Лемму 2.3}
(отождествление: $f_1 \leftrightarrow f(x_*)$,\; $f_2 \leftrightarrow \mathbf{y}$):

$$
f(x_*)\mid D
\;\sim\;
\mathrm{MVT}\!\left(
\underbrace{k_*^T K^{-1} \mathbf{y}}_{\mu_{1|2}},\;\;
\underbrace{\dfrac{\nu - 2 + \beta}{\nu - 2 + n}
\,\bigl(k_\theta(x_*,x_*) - k_*^T K^{-1} k_*\bigr)}_{K_{1|2}},\;\;
\underbrace{\nu + n}_{\nu_{1|2}}
\right),
$$

$$
\text{где}\quad \beta = \mathbf{y}^T K^{-1} \mathbf{y}.
$$

\vspace{3pt}

\begin{block}{Ключевое отличие от GP-регрессии}
\begin{tabular}{ll}
\textbf{GP:} & $\mathbb{V}[f(x_*)\mid\mathbf{y}]
  = k_\theta(x_*,x_*) - k_*^TK^{-1}k_*$
  \quad (не зависит от $\mathbf{y}$)\\[4pt]
\textbf{TP:} & $\mathbb{V}[f(x_*)\mid\mathbf{y}]
  = \dfrac{\nu-2+\beta}{\nu-2+n}\,
  \bigl(k_\theta(x_*,x_*) - k_*^TK^{-1}k_*\bigr)$
  \quad (растёт при выбросах)
\end{tabular}
\end{block}

}
\end{frame}

%----------------------------------------------------------------------------------------------------------

\begin{frame}{Функция потерь TP-регрессии}

{
\textbf{Отрицательное логарифмическое маргинальное правдоподобие:}

\footnotesize{
\[
\mathcal{L}(\theta) = \frac{n}{2}\log\!\bigl((\nu-2)\pi\bigr)
+ \frac{1}{2}\log|K|
- \log\Gamma\!\left(\frac{\nu+n}{2}\right)\
+ \log\Gamma\!\left(\frac{\nu}{2}\right)
+ \frac{\nu+n}{2}\log\!\left(1+\frac{\beta}{\nu-2}\right)
\]
}
где $\beta = (y-\mu)^T K^{-1}(y-\mu)$.

\vspace{5pt}

\textbf{Оцениваемые параметры:}
\begin{itemize}
    \item $\theta$ --- гиперпараметры ковариационной функции $k_\theta$
    \item $\sigma_n^2$ --- уровень шума
    \item $\nu$ --- степени свободы (тяжесть хвостов)
\end{itemize}

\vspace{5pt}

}

\end{frame}

\begin{frame}{Связь GP с фильтром Калмана}
\footnotesize

\begin{block}{\footnotesize Идея}
    Временной GP эквивалентен линейной динамической системе:
    $$
        f(t) \sim \mathrm{GP}(0,\, k(t,t'))
        \quad\Longleftrightarrow\quad
        \text{СДУ пространства состояний}
    $$
\end{block}

\begin{columns}[T]

    \column{0.48\textwidth}
    \begin{block}{\footnotesize Непрерывная форма (СДУ)}
        \vspace{2pt}
        \textbf{Начальное условие:}
        $$
            \mathbf{f}(0) \sim \mathcal{N}(0,\; P_\infty)
        $$
        \textbf{Уравнение состояния:}
        $$
            d\mathbf{f}(t) = F\mathbf{f}(t)\,dt + L\,dW(t)
        $$
        \textbf{Уравнение наблюдения:}
        $$
            y(t_k) = H\mathbf{f}(t_k) + \varepsilon_k
        $$
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{\footnotesize Дискретная форма}
        \vspace{2pt}
        \textbf{Начальное условие:}
        $$
            \mathbf{f}_0 \sim \mathcal{N}(0,\; P_\infty)
        $$
        \textbf{Переход состояния:}
        $$
            \mathbf{f}_k = A_{k-1}\mathbf{f}_{k-1} + \mathbf{q}_{k-1}
        $$
        $$
            \mathbf{q}_{k-1} \sim \mathcal{N}(0,\, Q_{k-1})
        $$
        \textbf{Наблюдение:}
        $$
            y_k = H\mathbf{f}_k + \varepsilon_k
        $$
    \end{block}

\end{columns}

\end{frame}


%──────────────────────────────────────────────────────────────────
\begin{frame}{Главный результат: TP как модель пространства состояний}
\footnotesize

\begin{block}{\footnotesize Теорема 3.2}
    Временной TP эквивалентен масштабированной смеси гауссовских СДУ:
    $$
        f(t) \sim \mathrm{TP}(0,\, k(t,t'), \nu)
        \quad\Longleftrightarrow\quad
        \text{гауссовская смесь СДУ пространства состояний}
    $$
\end{block}

\begin{columns}[T]

    \column{0.48\textwidth}
    \begin{block}{\footnotesize Непрерывная форма (СДУ)}
        \vspace{2pt}
        \textbf{Начальное условие:}
        $$
            \mathbf{f}(0) \sim \mathcal{N}(0,\; \textcolor{red}{\gamma}\, P_0)
        $$
        \textbf{Уравнение состояния:}
        $$
            d\mathbf{f}(t) = F\mathbf{f}(t)\,dt + L\,dW(t)
        $$
        $$
            W(t)\sim\mathcal{N}(0,\;\textcolor{red}{\gamma}\, Q_c)
        $$
        \textbf{Уравнение наблюдения:}
        $$
            y(t_k) = H\mathbf{f}(t_k) + \varepsilon_k
        $$
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{\footnotesize Дискретная форма}
        \vspace{2pt}
        \textbf{Начальное условие:}
        $$
            \mathbf{f}_0 \sim \mathcal{N}(0,\; \textcolor{red}{\gamma}\, P_0)
        $$
        \textbf{Переход состояния:}
        $$
            \mathbf{f}_k = A_{k-1}\mathbf{f}_{k-1} + \mathbf{q}_{k-1}
        $$
        $$
            \mathbf{q}_{k-1} \sim \mathcal{N}(0,\;\textcolor{red}{\gamma}\, Q_{k-1})
        $$
        \textbf{Наблюдение:}
        $$
            y_k = H\mathbf{f}_k + \varepsilon_k
        $$
    \end{block}

\end{columns}

\end{frame}

\begin{frame}{Алгоритм: фильтрация (вперёд)}
\footnotesize

\begin{columns}[T]

    % ─── Левая колонка: обозначения ───────────────────────────
    \column{0.28\textwidth}
    \begin{block}{\footnotesize Обозначения}
        \vspace{2pt}
        {\footnotesize
        \begin{tabular}{cl}
            $m_{k|k}$          & среднее          \\ [3pt]
            $P_{k|k}$          & ковариация        \\[3pt]
            $\nu_k$            & степени св.       \\[3pt]
            $\gamma_k$         & масштаб           \\[3pt]
            $v_k$              & инновация         \\[3pt]
            $S_k$              & дисп. иннов.      \\[3pt]
            $K_k$              & усиление          \\[3pt]
            $A_k(\theta)$      & переход           \\[3pt]
            $Q_k(\theta)$      & шум перехода      \\[3pt]
            $H_k(\theta)$      & наблюдение        \\
        \end{tabular}}
    \end{block}

    \vspace{3pt}

    \begin{block}{\footnotesize Инициализация}
        \vspace{2pt}
        {\footnotesize
        \begin{tabular}{ll}
            $\nu_0    = \nu$         & $\gamma_0 = 1$          \\[3pt]
            $m_{0|0} = 0$            & $P_{0|0} = P_0(\theta)$ \\
        \end{tabular}}
    \end{block}

    % ─── Правая колонка: алгоритм ─────────────────────────────
    \column{0.68\textwidth}
    \begin{block}{\footnotesize Шаги для $k = 1, \ldots, n$}
        \vspace{2pt}

        \textbf{1. Предсказание:}
        \vspace{-4pt}
        $$
            m_{k|k-1} = A_{k-1}(\theta)\,m_{k-1|k-1}
        $$
        \vspace{-14pt}
        $$
            P_{k|k-1} = A_{k-1}(\theta)\,P_{k-1|k-1}\,A_{k-1}(\theta)^T
            + \gamma_{k-1}\,Q_{k-1}(\theta)
        $$

        \textbf{2. Инновация:}
        \vspace{-4pt}
        $$
            v_k = y_k - H_k(\theta)\,m_{k|k-1}, \qquad
            S_k = H_k(\theta)\,P_{k|k-1}\,H_k(\theta)^T
        $$

        \textbf{3. Обновление масштаба:}
        \vspace{-4pt}
        $$
            \nu_k = \nu_{k-1} + 1
        $$
        \vspace{-14pt}
        $$
            \gamma_k = \frac{\nu_{k-1} - 2 + v_k^T S_k^{-1} v_k}{\nu_k - 2}
            \cdot \gamma_{k-1}
        $$

        \textbf{4. Обновление состояния:}
        \vspace{-4pt}
        $$
            K_k = P_{k|k-1}\,H_k(\theta)^T S_k^{-1}
        $$
        \vspace{-14pt}
        $$
            m_{k|k} = m_{k|k-1} + K_k\,v_k, \quad
            P_{k|k} = \frac{\gamma_{k}}{\gamma_{k-1}}
            \bigl(P_{k|k-1} - K_k\,S_k\,K_k^T\bigr)
        $$
    \end{block}

\end{columns}

\end{frame}

        \begin{tabular}{l}
            $m_{n|n},\; P_{n|n}$ \\[3pt]

    \end{block}

        \vspace{2pt}

        \textbf{1. Предсказание:}
        \vspace{-4pt}
        $$
            m_{k+1|k} = A_{k}(\theta)\,m_{k|k}
        $$
        \vspace{-14pt}
        $$
            P_{k+1|k} = A_{k}(\theta)\,P_{k|k}\,A_{k}(\theta)^T
            + \gamma_{k}\,Q_{k}(\theta)
        $$

        \textbf{2. Коэффициент сглаживания:}
        \vspace{-4pt}
        $$
            G_k = P_{k|k}\,A_k(\theta)^T\,P_{k+1|k}^{-1}
        $$

        \textbf{3. Обновление среднего:}
        \vspace{-4pt}
        $$
            m_{k|n} = m_{k|k} + G_k\!\left(m_{k+1|n} - m_{k+1|k}\right)
        $$

        \textbf{4. Обновление ковариации:}
        \vspace{-4pt}
        $$
            P_{k|n} = \frac{\gamma_n}{\gamma_{k-1}}
            \bigl(P_{k|k} - G_k\,P_{k+1|k}\,G_k^T\bigr)
            + G_k\,P_{k+1|n}\,G_k^T
        $$

    \end{block}

\end{columns}

\end{frame}

\begin{frame}{Маргинальное правдоподобие}
\footnotesize



% ─── Основная формула ─────────────────────────────────────────
\begin{block}{\footnotesize Отрицательное лог-правдоподобие}
$$
    \mathcal{L}(\theta) = \sum_{k=1}^{n} \Bigl[
        \tfrac{1}{2}\log\bigl((\nu-2)\pi\bigr)
        + \tfrac{1}{2}\log|S_k(\theta)|
        + \log\Gamma\!\left(\tfrac{\nu_{k-1}}{2}\right)
        - \log\Gamma\!\left(\tfrac{\nu_k}{2}\right)
$$
$$
        + {\frac{1}{2}\frac{\log(\nu{k-1}-2)}{\nu-2}} + \tfrac{\nu_k}{2}
        \log\!\left(
            1 + \frac{v_k(\theta)^T S_k(\theta)^{-1} v_k(\theta)}{\nu_{k-1}-2}
        \right)
    \Bigr]
$$
\end{block}

\begin{block}{Обновление и вычислительная выгода}
Обновление происходит после каждой итерации алгоритма фильтрации, оно происходит до сходимости. При $m \ll n$: $O(nm^3) \approx O(n)$ против $O(n^3)$ для наивного решения.\\
\end{block}


\end{frame}
