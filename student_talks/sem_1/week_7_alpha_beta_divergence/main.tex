\begin{frame}{Motivation}

\textbf{Problem:} Standard VI with KL-divergence has limitations

\begin{itemize}
    \item Mode-seeking behavior (underestimates variance)
    \item Sensitive to outliers in data
\end{itemize}

\vspace{0.3cm}

\textbf{Existing solutions} address only one issue:
\begin{itemize}
    \item Rényi divergence: controls mass-covering
    \item Gamma divergence: controls robustness
\end{itemize}

\vspace{0.3cm}

\textbf{Goal:} Unified framework for both properties

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{Variational Inference Recap}

\textbf{Posterior approximation:} $p(\theta|X) \approx q(\theta)$

\vspace{0.3cm}

\textbf{Standard objective:} minimize $D[q(\theta)\|p(\theta|X)]$

\vspace{0.3cm}

\textbf{ELBO formulation:}
\[
\mathcal{L}_D(q, X, \varphi) = \log p(X|\varphi) - D(q(\theta)\|p(\theta|X, \varphi))
\]

\vspace{0.3cm}

\textbf{Challenge:} Choice of divergence $D$ affects approximation quality
\begin{itemize}
    \item Mass-covering vs mode-seeking
    \item Robustness to outliers
    \item Computational tractability
\end{itemize}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{Problems with KL Divergence}

\textbf{Formula:}
\[
D_{\text{KL}}(q\|p) = \int q(\theta) \log\frac{q(\theta)}{p(\theta)} d\theta
\]

\textbf{Gradient:}
\[
\frac{d}{d\varphi} D_{\text{KL}}(q\|p) = -\int \frac{dq(\theta)}{d\varphi} \left[\log\frac{p(\theta)}{q(\theta)} - 1\right] d\theta
\]

\vspace{0.3cm}

\textbf{Key issue:} Log-term increases cost of overestimating $p$

\vspace{0.2cm}

\textbf{Result:} 
\begin{itemize}
    \item Underestimation of posterior variance (Turner \& Sahani, 2011)
    \item Mode-seeking behavior: focuses on single mode
    \item Not robust to outliers
\end{itemize}

\end{frame}
%-----------------------------------------------------------------------------------------------------

\begin{frame}{Renyi and Gamma divergences}

\[
D_{\alpha}(q \Vert p) = \frac{1}{\alpha - 1} \log \int q(\theta)^\alpha\, p(\theta)^{1 - \alpha} \, d\theta
\]


\begin{multline*}
    D_{\gamma}^{\beta}(q \Vert p) = \frac{1}{\beta(\beta+1)} \log \int p(\theta)^{\beta + 1} d\theta \\
    - \frac{1}{\beta} \log \int p(\theta)q(\theta)^{\beta} d\theta \\
    + \frac{1}{\beta+1} \log \int q(\theta)^{1+\beta} d\theta
\end{multline*}


\end{frame}

%-----------------------------------------------------------------------------------------------------
\begin{frame}{Scale Invariant Alpha-Beta (sAB) Divergence}

\textbf{Formula:}
\begin{multline*}
D^{\alpha,\beta}_{\text{sAB}}(p\|q) = \frac{1}{\beta(\alpha+\beta)} \log \int p(\theta)^{\alpha+\beta} d\theta \\
+ \frac{1}{\alpha(\alpha+\beta)} \log \int q(\theta)^{\alpha+\beta} d\theta \\
- \frac{1}{\alpha\beta} \log \int p(\theta)^{\alpha} q(\theta)^{\beta} d\theta
\end{multline*}

\vspace{0.3cm}

\textbf{Two control parameters:} $\alpha$ and $\beta$
\begin{itemize}
    \item Extended to entire $\mathbb{R}^2$ domain
    \item Smooth interpolation between divergences
\end{itemize}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{Special Cases of sAB Divergence}

\begin{table}[]
\centering
\begin{tabular}{|l|c|}
\hline
\textbf{Divergence} & \textbf{Parameters} \\
\hline
KL divergence & $(\alpha, \beta) = (0, 1)$ \\
\hline
Reverse KL & $(\alpha, \beta) = (1, 0)$ \\
\hline
Rényi divergence & $\alpha + \beta = 1$ \\
\hline
Gamma divergence & $\alpha = 1$ \\
\hline
Hellinger distance & $(\alpha, \beta) = (0.5, 0.5)$ \\
\hline
Chi-square divergence & $(\alpha, \beta) = (2, -1)$ \\
\hline
\end{tabular}
\end{table}

\vspace{0.3cm}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{Control Parameters: $\alpha$ and $\beta$}

Use $\lambda = \alpha + \beta$ and $\beta$ for interpretation

\vspace{0.3cm}

\textbf{$\lambda$ (robustness parameter):}
\begin{itemize}
    \item $\lambda < 2$: robust to outliers
    \item $\lambda > 2$: focuses on outliers
\end{itemize}

\vspace{0.3cm}

\textbf{$\beta$ (mass-covering parameter):}
\begin{itemize}
    \item $\beta > 1$: mass-covering behavior
    \item $\beta < 1$: mode-seeking behavior
\end{itemize}

\vspace{0.3cm}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{sAB Divergence Gradient}

\begin{multline*}
\frac{d}{d\varphi} D^{\alpha,\beta}_{\text{sAB}}(q\|p) = -\frac{1}{\beta} \Bigg[ \frac{\int \frac{dq(\theta)}{d\varphi} q(\theta)^{\alpha+\beta-1} \left(\frac{p(\theta)}{q(\theta)}\right)^{\beta} d\theta}{\int q(\theta)^{\alpha} p(\theta)^{\beta} d\theta} \\
- \frac{\alpha}{\beta} \frac{\int \frac{dq(\theta)}{d\varphi} q(\theta)^{\alpha+\beta-1} d\theta}{\int q(\theta)^{\alpha+\beta} d\theta} \Bigg]
\end{multline*}

\vspace{0.3cm}

\textbf{Key terms:}
\begin{itemize}
    \item $q(\theta)^{\alpha+\beta-1}$: controls robustness
    \item $\left(\frac{p(\theta)}{q(\theta)}\right)^{\beta}$: controls mass-covering
\end{itemize}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{sAB Variational Objective}

\textbf{Direct divergence optimization} (not ELBO!)

\begin{multline*}
D^{\alpha,\beta}_{\text{sAB}}(q(\theta)\|p(\theta|X)) = \frac{1}{\alpha(\alpha+\beta)} \log \mathbb{E}_q\left[\frac{p(\theta,X)^{\alpha+\beta}}{q(\theta)^{\alpha+\beta}}\right] \\
+ \frac{1}{\beta(\alpha+\beta)} \log \mathbb{E}_q\left[q(\theta)^{\alpha+\beta-1}\right] \\
- \frac{1}{\alpha\beta} \log \mathbb{E}_q\left[q(\theta)^{\alpha+\beta-1}\left(\frac{p(\theta,X)}{q(\theta)}\right)^{\beta}\right]
\end{multline*}

\vspace{0.3cm}

\begin{itemize}
    \item All terms are expectations w.r.t. $q$
    \item $p(X)$ terms cancel out
    \item Monte Carlo approximation with $K$ samples
\end{itemize}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{Optimization Framework}

\textbf{Monte Carlo approximation:}
\begin{itemize}
    \item Sample $\theta_1, \ldots, \theta_K \sim q(\theta)$
    \item Approximate each expectation term
    \item Use $K \approx 100$ samples
\end{itemize}

\vspace{0.3cm}

\textbf{Gradient computation:}
\begin{itemize}
    \item Reparameterization trick for differentiability
    \item Can leverage modern VI optimization methods
    \item ADAM optimizer with learning rate 0.01
\end{itemize}

\vspace{0.3cm}

\end{frame}
%-----------------------------------------------------------------------------------------------------
\begin{frame}{Experimental Setup}

\textbf{Two main experiments:}

\begin{enumerate}
    \item \textbf{Synthetic data:} Bayesian linear regression with 5\% corrupted data
    \item \textbf{UCI datasets:} Bayesian neural networks with artificial outliers
\end{enumerate}

\vspace{0.3cm}

\textbf{Network architecture:}
\begin{itemize}
    \item 2-layer with 50 hidden units
    \item ReLU activation
    \item Fully factorized Gaussian approximation
\end{itemize}

\vspace{0.3cm}

\textbf{Parameter selection:}
\begin{itemize}
    \item Cross-validation to select optimal $(\alpha, \beta)$
    \item Compare with KL, Rényi, and Gamma divergences
\end{itemize}

\end{frame}

%-----------------------------------------------------------------------------------------------------
\begin{frame}{Key Insights}

\begin{enumerate}
    \item \textbf{Unified framework:} mass-covering + robustness in one objective
    \item \textbf{Direct divergence optimization} (not ELBO)
    \item \textbf{Smooth interpolation} between known divergences
    \item \textbf{Two interpretable parameters} with clear effects:
    \begin{itemize}
        \item $\lambda$ controls outlier sensitivity
        \item $\beta$ controls mode-seeking vs mass-covering
    \end{itemize}
    \item \textbf{Cross-validation} can find optimal parameters
    \item \textbf{Most effective} when $\lambda < 2$ and $\beta < 1$
\end{enumerate}

\end{frame}
