\begin{frame}{Motivation: Standard SGD Training}

Standard approach: SGD with decaying learning rate until convergence

\vspace{0.5cm}

\begin{block}{Key Observations}
\begin{itemize}
    \item Local optima found by SGD can be connected by curves of constant loss
    \item There exist regions of high-performing networks, not just isolated points
    \item But where does SGD converge within these regions?
\end{itemize}
\end{block}

\vspace{0.3cm}

\begin{alertblock}{Core Problem}
SGD converges to points on the periphery of optimal regions.\\
Can we find more central, flatter solutions?
\end{alertblock}

\end{frame}

\begin{frame}{Loss Surface Geometry}

\begin{block}{SGD with Constant Learning Rate}
Running SGD with constant learning rate $\alpha$ approximately samples:
\begin{equation*}
w_i \sim \mathcal{N}(\hat{w}, \Sigma(\alpha))
\end{equation*}
\end{block}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Standard SGD:}
\begin{itemize}
    \item Decaying LR picks one point on surface
    \item Stops at periphery
    \item Near directions of sharp ascent
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Key Insight:}
\begin{itemize}
    \item Train loss $\neq$ test error surfaces
    \item They are shifted
    \item Central points generalize better
\end{itemize}
\end{column}
\end{columns}

\vspace{0.3cm}

\begin{block}{Solution}
Average multiple points $\rightarrow$ move to the center!
\end{block}

\end{frame}

\begin{frame}{Stochastic Weight Averaging (SWA)}

\begin{block}{The Algorithm}
Collect weight vectors $\{w_1, w_2, \ldots, w_n\}$ along SGD trajectory:
\begin{equation*}
w_{\text{SWA}} = \frac{1}{n} \sum_{i=1}^{n} w_i
\end{equation*}
\end{block}

\begin{enumerate}
    \item Train with standard SGD for $\sim$75\% of budget $\rightarrow$ get $\hat{w}$
    \item Continue with cyclical or constant learning rate
    \item Periodically save weights and update running average:
    \begin{equation*}
    w_{\text{SWA}} \leftarrow \frac{w_{\text{SWA}} \cdot n + w}{n + 1}
    \end{equation*}
    \item Recompute batch norm statistics for $w_{\text{SWA}}$
\end{enumerate}

\vspace{0.2cm}

\begin{block}{Computational Cost}
Memory: +10\% \quad Time: +0-5\%
\end{block}

\end{frame}

\begin{frame}{Learning Rate Schedules}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Cyclical Schedule}

Linearly decrease from $\alpha_1$ to $\alpha_2$ in cycles:
\begin{equation*}
\alpha(i) = (1 - t(i))\alpha_1 + t(i)\alpha_2
\end{equation*}
\begin{equation*}
t(i) = \frac{1}{c}(\text{mod}(i-1, c) + 1)
\end{equation*}

\vspace{0.2cm}
\small
At cycle end: capture weights, jump back to $\alpha_1$
\normalsize
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Constant Schedule}

Simply set $\alpha(i) = \alpha_1$ for all iterations

\vspace{0.2cm}
\small
Capture weights every epoch
\normalsize
\end{column}
\end{columns}

\vspace{0.5cm}

\begin{block}{Hyperparameter Guidelines}
\begin{itemize}
    \item $\alpha_1$: between max and min LR from standard training (0.01-0.05)
    \item Cycle length $c$: 3-5 epochs works well
    \item Method is robust to hyperparameter choices
\end{itemize}
\end{block}

\end{frame}

\begin{frame}{Why Does SWA Work?}

\begin{block}{1. Geometric Interpretation}
SGD with cyclical/constant LR traverses surface of optimal set\\
$\rightarrow$ Averaging moves inside to more central points
\end{block}

\begin{block}{2. Connection to Ensembling}
Consider network predictions $f(\cdot)$ and Taylor expansion:
\begin{equation*}
f(w_i) = f(w_{\text{SWA}}) + \langle \nabla f(w_{\text{SWA}}), \Delta_i \rangle + O(\|\Delta_i\|^2)
\end{equation*}
where $\Delta_i = w_i - w_{\text{SWA}}$ and $\sum_i \Delta_i = 0$. Thus:
\begin{equation*}
\bar{f} - f(w_{\text{SWA}}) = O(\Delta^2)
\end{equation*}
SWA approximates ensemble to second order with single model!
\end{block}

\begin{block}{3. Solution Width}
SWA finds demonstrably flatter solutions than SGD
\end{block}

\end{frame}

\begin{frame}{Key Insights}

\begin{block}{1. Geometry Matters}
\begin{itemize}
    \item Loss surfaces contain regions of good solutions
    \item SGD explores periphery; SWA finds center
    \item Train loss $\neq$ test error (shifted surfaces)
\end{itemize}
\end{block}

\begin{block}{2. Solution Width}
\begin{itemize}
    \item SWA solutions are wider/flatter than SGD
    \item Robust to perturbations $\rightarrow$ better generalization
    \item $w_{\text{SWA}}$: slightly higher train loss, lower test error
\end{itemize}
\end{block}

\begin{block}{3. Practical Benefits}
\begin{itemize}
    \item Trivial to implement ($<$20 lines)
    \item Negligible overhead (0-5\%)
    \item Universal: works with any architecture, optimizer, dataset
    \item Single model with ensemble-level performance
\end{itemize}
\end{block}

\end{frame}

\begin{frame}{Conclusion}

\begin{alertblock}{Stochastic Weight Averaging}
Simple technique with profound geometric insight:\\
Average weights along SGD trajectory with cyclical/constant LR
\end{alertblock}

\textbf{Key Contributions:}
\begin{itemize}
    \item 0.5-1.5\% test accuracy improvement across architectures
    \item Single model matches/exceeds ensemble performance
    \item Demonstrably wider optima than standard SGD
    \item Essentially no computational overhead
\end{itemize}
