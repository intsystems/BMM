% ========== SECTION 1: INTRODUCTION ==========
\section{Introduction}

\begin{frame}{The Limitations of MLPs}
\begin{itemize}
    \item Multi-Layer Perceptrons (MLPs) are foundational but have significant drawbacks:
    \begin{itemize}
        \item Fixed activation functions on \textbf{nodes} (neurons)
        \item Linear transformations $\mathbf{W}$ separate from nonlinearities $\sigma$
        \item Consume majority of parameters in transformers
        \item Less interpretable without post-analysis tools
    \end{itemize}
    
    \item MLP formulation:
    \[
    \mathrm{MLP}(\mathbf{x}) = (\mathbf{W}_{L-1} \circ \sigma \circ \mathbf{W}_{L-2} \circ \sigma \circ \cdots \circ \mathbf{W}_1 \circ \sigma \circ \mathbf{W}_0)\mathbf{x}
    \]
\end{itemize}
\end{frame}

\begin{frame}{Kolmogorov-Arnold Representation Theorem}
\begin{theorem}[Kolmogorov-Arnold, 1956-1957]
If $f:[0,1]^n \rightarrow \mathbb{R}$ is a multivariate continuous function, then $f$ can be written as a finite composition of continuous functions of a single variable and the binary operation of addition:
\[
f(\mathbf{x}) = f(x_1, \cdots, x_n) = \sum_{q=1}^{2n+1} \Phi_q\left( \sum_{p=1}^n \phi_{q,p}(x_p) \right)
\]
where $\phi_{q,p}:[0,1]\rightarrow \mathbb{R}$ and $\Phi_q:\mathbb{R}\rightarrow \mathbb{R}$.
\end{theorem}

\begin{itemize}
    \item Kolmogorov and Arnold showed that the only true multivariate function is the sum, since every other function can be written using univariate functions and summing.
    \item Previously considered impractical due to non-smooth inner functions
\end{itemize}
\end{frame}

\begin{frame}{KANs: A New Paradigm}
\begin{itemize}
    \item \textbf{Kolmogorov-Arnold Networks (KANs)} leverage the theorem with modern deep learning
    \item Key differences from MLPs:
    \begin{itemize}
        \item Learnable activation functions on \textbf{edges} (weights)
        \item No linear weight matrices
        \item Each "weight" is a univariate function parametrized as a spline
        \item Nodes perform simple summation
    \end{itemize}
    
    \item Combines strengths of splines (accuracy in 1D) and MLPs (compositional structure learning)
\end{itemize}
\end{frame}

% ========== SECTION 2: KAN ARCHITECTURE ==========
\section{KAN Architecture}

\begin{frame}{Mathematical Formulation}
\begin{itemize}
    \item KAN layer with $n_{\text{in}}$ inputs and $n_{\text{out}}$ outputs:
    \[
    \bm{\Phi} = \{\phi_{q,p}\}, \quad p=1,\dots,n_{\text{in}}, \quad q=1,\dots,n_{\text{out}}
    \]
    
    \item Forward pass for layer $l$:
    \[
    x_{l+1,j} = \sum_{i=1}^{n_l} \phi_{l,j,i}(x_{l,i}), \quad j=1,\dots,n_{l+1}
    \]
    
    \item Matrix form:
    \[
    \mathbf{x}_{l+1} = 
    \begin{pmatrix}
    \phi_{l,1,1}(\cdot) & \cdots & \phi_{l,1,n_l}(\cdot) \\
    \vdots & \ddots & \vdots \\
    \phi_{l,n_{l+1},1}(\cdot) & \cdots & \phi_{l,n_{l+1},n_l}(\cdot)
    \end{pmatrix}
    \mathbf{x}_l
    \]
\end{itemize}
\end{frame}

\begin{frame}{Activation Function Parametrization}
\begin{itemize}
    \item Each activation function $\phi(x)$ is parametrized as:
    \[
    \phi(x) = w_b b(x) + w_s \cdot \text{spline}(x)
    \]
    
    \item Base function: $b(x) = \text{silu}(x) = x/(1+e^{-x})$
    
    \item Spline component: $\text{spline}(x) = \sum_i c_i B_i(x)$
    \begin{itemize}
        \item $B_i(x)$: B-spline basis functions
        \item $c_i$: Trainable coefficients
    \end{itemize}
    
    \item Grid extension: Refine spline grids without retraining from scratch
\end{itemize}
\end{frame}

\begin{frame}{Approximation Theory for KANs}
\begin{theorem}[Approximation Theory, KAT]
Let $\mathbf{x} = (x_1, \dots, x_n)$. Suppose $f(\mathbf{x})$ admits representation
\[
f = (\bm{\Phi}_{L-1} \circ \bm{\Phi}_{L-2} \circ \cdots \circ \bm{\Phi}_1 \circ \bm{\Phi}_0)\mathbf{x}
\]
with $(k+1)$-times continuously differentiable $\Phi_{l,i,j}$. Then there exists constant $C$ such that:
\[
\|f - (\bm{\Phi}_{L-1}^G \circ \cdots \circ \bm{\Phi}_0^G)\mathbf{x}\|_{C^m} \leq CG^{-k-1+m}
\]
for any $0 \leq m \leq k$.
\end{theorem}

\begin{itemize}
    \item \textbf{Breaks curse of dimensionality}: Error rate independent of dimension
    \item Scaling exponent: $\alpha = k+1$ (e.g., $\alpha = 4$ for cubic splines)
\end{itemize}
\end{frame}

% ========== SECTION 3: KANs ARE ACCURATE ==========
\section{KANs are Accurate}

\begin{frame}{Neural Scaling Laws}
\begin{itemize}
    \item Test loss $\ell$ scales with parameters $N$ as $\ell \propto N^{-\alpha}$
    \item Comparison of scaling exponents:
    \begin{itemize}
        \item \textbf{Sharma \& Kaplan}: $\alpha = (k+1)/d$ (suffers from COD)
        \item \textbf{Michaud et al.}: $\alpha = (k+1)/2$ (max arity 2)
        \item \textbf{Poggio et al.}: $\alpha = m/2$ (compositional sparsity)
        \item \textbf{KANs}: $\alpha = k+1$ (e.g., $\alpha = 4$ for cubic splines)
    \end{itemize}
    
    \item KANs achieve the \textbf{best possible scaling exponent}
\end{itemize}

\end{frame}

\begin{frame}{PDE Solving: Poisson Equation}
\begin{itemize}
    \item Consider Poisson equation on $\Omega = [-1,1]^2$:
    \[
    u_{xx} + u_{yy} = f \quad \text{in } \Omega, \quad u = 0 \quad \text{on } \partial\Omega
    \]
    
    \item Physics-Informed Neural Networks (PINNs) framework:
    \[
    \text{loss}_{\text{pde}} = \alpha \text{loss}_i + \text{loss}_b
    \]
    where
    \[
    \text{loss}_i = \frac{1}{n_i} \sum_{i=1}^{n_i} |u_{xx}(z_i) + u_{yy}(z_i) - f(z_i)|^2
    \]
    \[
    \text{loss}_b = \frac{1}{n_b} \sum_{i=1}^{n_b} u^2
    \]
\end{itemize}
\end{frame}

% ========== SECTION 4: KANs ARE INTERPRETABLE ==========
\section{KANs are Interpretable}

\begin{frame}{Simplification Techniques}
\begin{itemize}
    \item \textbf{Sparsification}: L1 + entropy regularization
    \[
    \ell_{\text{total}} = \ell_{\text{pred}} + \lambda\left(\mu_1 \sum_{l=0}^{L-1} |\bm{\Phi}_l|_1 + \mu_2 \sum_{l=0}^{L-1} S(\bm{\Phi}_l)\right)
    \]
    where $|\phi|_1 = \frac{1}{N_p} \sum_{s=1}^{N_p} |\phi(x^{(s)})|$
    
    \item \textbf{Pruning}: Remove nodes with low incoming/outgoing scores
    \[
    I_{l,i} = \max_k(|\phi_{l-1,i,k}|_1), \quad O_{l,i} = \max_j(|\phi_{l+1,j,i}|_1)
    \]
    
    \item \textbf{Symbolification}: Fix activation functions to symbolic forms
    \[
    \text{fix\_symbolic}(l,i,j,f)
    \]
\end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Mathematical Discussion}
\begin{itemize}
    \item \textbf{Generalized Kolmogorov-Arnold Theorem}: 
    \begin{itemize}
        \item Deeper representations may enable smoother activations
        \item "Kolmogorov-Arnold depth" as function class characterization
    \end{itemize}
    
    \item \textbf{Algorithmic Improvements}:
    \begin{itemize}
        \item Alternative parametrizations (radial basis functions)
        \item Multi-head activation grouping
        \item Multi-level training strategies
    \end{itemize}
    
    \item \textbf{Comparison with UAT}:
    \begin{itemize}
        \item UAT: No error bounds, suffers from COD
        \item KAT: Quantifies error, breaks COD for compositional functions
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Future Directions \& Applications}
\begin{itemize}
    \item \textbf{Mathematical Foundation}:
    \begin{itemize}
        \item Deeper Kolmogorov-Arnold representations
        \item Connection to function space theory
    \end{itemize}
    
    \item \textbf{Scientific Applications}:
    \begin{itemize}
        \item Navier-Stokes equations, density functional theory
        \item "Kansformers": Replace MLPs with KANs in transformers
    \end{itemize}
    
    \item \textbf{AI + Science Paradigm}:
    \begin{itemize}
        \item KANs as "language" for scientist-AI collaboration
        \item Interpretable function-based communication
    \end{itemize}
\end{itemize}

\begin{block}{When to Use KANs vs MLPs}
\begin{itemize}
    \item \textbf{Use KANs}: Interpretability and accuracy critical, training speed secondary
    \item \textbf{Use MLPs}: Fast training required, interpretability less important
\end{itemize}
\end{block}
\end{frame}
